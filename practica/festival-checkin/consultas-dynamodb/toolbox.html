GetItemCommand

Leer un evento

Evento con ID 1

src/eventos/buscar-evento/repository.mjs

import { GetItemCommand } from 'dynamodb-toolbox/entity/actions/get';
import { MODELO_EVENTO } from '../models/evento.model.mjs';

export async function buscarEventoRepository(eventId) {
  const { Item } = await MODELO_EVENTO.build(GetItemCommand)
    .key({ eventId })
    .options({ consistent: true })
    .send();

  return Item ?? null;
}

PutItemCommand

Crear un evento

Añadir el concierto del aula

src/eventos/crear-evento/repository.mjs

import { PutItemCommand } from 'dynamodb-toolbox/entity/actions/put';
import { MODELO_EVENTO } from '../models/evento.model.mjs';

export async function crearEventoRepository(eventId, title) {
  const evento = {
    eventId,
    title
  };

  const comando = MODELO_EVENTO.build(PutItemCommand)
    .item(evento)
    .options({
      condition: { attr: 'eventId', exists: false }
    });

  await comando.send();
}
Abrir este archivo en el repositorio del proyecto
  1. JavaScripteventId + title
  2. DynamoDBEVENT_ID + TITLE
  3. ResultadoSe guarda el evento
.item(evento)
Entrega la ficha completa. El modelo cambia los nombres de JavaScript a los de DynamoDB.
exists: false
El ID debe estar libre. DynamoDB evalúa esta condición al escribir y conserva el evento anterior si el ID ya existe.
.send()
Ejecuta la escritura. Si la condición falla, la promesa se rechaza.

Personas inscritas en el evento 1

Practica en una carpeta nueva desde la raíz de festival-checkin-01, con la sesión y las tablas de tu equipo. Crea y abre el archivo antes de copiar el contenido.

vi src/inscripciones/listar-inscritos/repository.mjs
import { QueryCommand } from 'dynamodb-toolbox/table/actions/query';
import { MODELO_INSCRIPCION } from '../models/inscripcion.model.mjs';

const INSCRIPCIONES_POR_PAGINA = 25;

export async function listarInscritosRepository(eventId, cursor) {
  const { Items = [], LastEvaluatedKey } = await MODELO_INSCRIPCION.table
    .build(QueryCommand)
    .entities(MODELO_INSCRIPCION)
    .query({ partition: eventId })
    .options({
      limit: INSCRIPCIONES_POR_PAGINA,
      exclusiveStartKey: cursor
    })
    .send();

  return { inscripciones: Items, siguienteCursor: LastEvaluatedKey ?? null };
}
  1. JavaScriptpartition: eventId
  2. DynamoDBEVENT_ID = 1
  3. ResultadoItems → pases del evento
MODELO_INSCRIPCION.table.build
Construye la consulta desde la tabla.
.entities(MODELO_INSCRIPCION)
Usa el modelo para traducir las fichas obtenidas a eventId, userId, registeredAt y checkedInAt cuando exista.
.query({ partition: eventId })
Elige el grupo EVENT_ID. Dentro de ese grupo los ítems salen ordenados por USER_ID.
LastEvaluatedKey
Pasa ese objeto completo como cursor si quieres continuar desde la última clave evaluada.
ScanCommand

Listar el catálogo de eventos

Recorrer una página del catálogo

src/eventos/listar-eventos/repository.mjs

import { ScanCommand } from 'dynamodb-toolbox/table/actions/scan';
import { MODELO_EVENTO } from '../models/evento.model.mjs';

const EVENTOS_POR_PAGINA = 25;

export async function listarEventosRepository(cursor) {
  const inicio = cursor == null ? undefined : { EVENT_ID: BigInt(cursor) };

  const { Items = [], LastEvaluatedKey } = await MODELO_EVENTO.table.build(ScanCommand)
    .entities(MODELO_EVENTO)
    .options({
      consistent: true,
      limit: EVENTOS_POR_PAGINA,
      exclusiveStartKey: inicio
    })
    .send();

  return {
    eventos: Items.map(({ eventId, title }) => ({ eventId: String(eventId), title })),
    siguienteCursor: LastEvaluatedKey ? String(LastEvaluatedKey.EVENT_ID) : null
  };
}
Abrir este archivo en el repositorio del proyecto
  1. JavaScriptcursor opcional
  2. DynamoDBHasta 25 ítems evaluados
  3. ResultadoItems → una página del catálogo
.table.build(ScanCommand)
Inicia un recorrido de la tabla.
.entities(MODELO_EVENTO)
Da formato a cada ítem del catálogo según Evento.
exclusiveStartKey
Recibe la clave desde la que continuar. En esta tabla se puede reconstruir con solo EVENT_ID, que es su clave completa.
String(eventId)
Convierte el ID a cadena para devolverlo en JSON. El cursor es null si DynamoDB omite LastEvaluatedKey.

import { UpdateItemCommand } from 'dynamodb-toolbox/entity/actions/update';
import { MODELO_INSCRIPCION } from '../models/inscripcion.model.mjs';

export async function registrarEntradaRepository(eventId, userId, checkedInAt) {
  const { Attributes } = await MODELO_INSCRIPCION.build(UpdateItemCommand)
    .item({ eventId, userId, checkedInAt })
    .options({
      condition: { attr: 'eventId', exists: true },
      returnValues: 'ALL_NEW'
    })
    .send();

  return Attributes;
}
  1. JavaScripteventId + userId + checkedInAt
  2. DynamoDBClaves del pase + CHECKED_IN_AT
  3. ResultadoAttributes → pase actualizado
.item({ eventId, userId, checkedInAt })
Los dos IDs localizan el pase. checkedInAt es el único atributo que cambias. Pasa una fecha ISO, por ejemplo 2026-11-12T09:12:00Z.
exists: true
Exige un pase existente para evitar que DynamoDB cree un ítem incompleto al actualizar.
returnValues: 'ALL_NEW'
Pide el ítem completo tras el cambio. Toolbox lo entrega en Attributes con los nombres del modelo.

De JavaScript a DynamoDB

new Table fija el nombre, el cliente y las claves físicas. new Entity describe la ficha. .key() identifica sus claves y .savedAs() traduce cada nombre. entityAttribute: false y timestamps: false evitan añadir los metadatos automáticos de Toolbox.

Evento

eventId → EVENT_ID · title → TITLE

La clave es numérica. La opción big permite usar BigInt.

src/eventos/models/evento.model.mjs

import { Entity } from 'dynamodb-toolbox/entity';
import { Table } from 'dynamodb-toolbox/table';
import { item } from 'dynamodb-toolbox/schema/item';
import { string } from 'dynamodb-toolbox/schema/string';
import { number } from 'dynamodb-toolbox/schema/number';
import { db, TABLA_EVENTOS_FCI } from '../../datos/dynamo.mjs';

const TABLA_MODELO_EVENTO = new Table({
  name: TABLA_EVENTOS_FCI,
  documentClient: db,
  partitionKey: { name: 'EVENT_ID', type: 'number' }
});

export const MODELO_EVENTO = new Entity({
  name: 'Evento',
  table: TABLA_MODELO_EVENTO,
  entityAttribute: false,
  timestamps: false,
  schema: item({
    eventId: number({ big: true }).key().savedAs('EVENT_ID'),
    title: string().savedAs('TITLE')
  })
});

Inscripción

eventId → EVENT_ID · userId → USER_ID

La tabla usa los dos identificadores. El modelo también traduce las fechas de inscripción y entrada.

src/inscripciones/models/inscripcion.model.mjs

import { Entity } from 'dynamodb-toolbox/entity';
import { Table } from 'dynamodb-toolbox/table';
import { item } from 'dynamodb-toolbox/schema/item';
import { number } from 'dynamodb-toolbox/schema/number';
import { string } from 'dynamodb-toolbox/schema/string';
import { db, TABLA_INSCRIPCIONES_FCI } from '../../datos/dynamo.mjs';

const tablaInscripciones = new Table({
  name: TABLA_INSCRIPCIONES_FCI,
  documentClient: db,
  partitionKey: { name: 'EVENT_ID', type: 'number' },
  sortKey: { name: 'USER_ID', type: 'string' }
});

export const MODELO_INSCRIPCION = new Entity({
  name: 'Inscripcion',
  table: tablaInscripciones,
  entityAttribute: false,
  timestamps: false,
  schema: item({
    eventId: number({ big: true }).key().savedAs('EVENT_ID'),
    userId: string().key().savedAs('USER_ID'),
    registeredAt: string().savedAs('REGISTERED_AT'),
    checkedInAt: string().optional().savedAs('CHECKED_IN_AT')
  })
});
